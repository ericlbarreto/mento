# Whiteboard back-end, Plan 2: voice (Deepgram STT, ElevenLabs TTS, audio endpoint, browser client)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Natural, low-latency voice: the browser streams the student's speech to Deepgram and gets live word captions; the server synthesizes the tutor's speech with ElevenLabs the moment a `speak` action exists, caches it, serves it as chunked PCM, and returns word timings for captions.

**Architecture:** `packages/voice/server` holds provider code behind small interfaces (`TtsProvider`, `mintSttToken`); `TtsService` runs one job per `speak` (start early, stream to the client, cache by hash). `packages/voice/client` holds browser code (Deepgram session, PCM player, microphone). `apps/api` exposes `POST /api/voice/token` and `GET /api/tts/:action_id`. Everything is testable with fakes; a smoke script measures the real providers (Day-0 checks).

**Tech Stack:** TypeScript, Node 24 (`fetch`), `ws` (ElevenLabs WebSocket needs an auth header), Web Audio API (AudioWorklet), React 19 (hook only), Vitest 5.

**Spec:** `docs/superpowers/specs/2026-10-30-whiteboard-backend-design.md` (§3, §5). Prerequisite: Plan 1 (contracts `WordTiming`, `SttCaption`, `SpeechPlayer`).

## Global Constraints

- Packages never read environment variables; only `apps/api/src/config.ts` does and passes plain options.
- Tutor audio format: PCM signed 16-bit little-endian, 24 kHz, mono (`output_format=pcm_24000`; contract field `audio_format: "pcm_s16le_24000"`).
- ElevenLabs model default `eleven_flash_v2_5`; Portuguese via `language_code=pt`. Deepgram model `nova-3`, `language=pt-BR`, `interim_results=true`, `endpointing=300`, `utterance_end_ms=1000`, `vad_events=true`, `smart_format=true`, `punctuate=true`; browser audio is linear16, 16 kHz, mono, 80 ms chunks (1280 samples).
- The Deepgram API key and the ElevenLabs API key never reach the browser; the browser gets only a short-lived Deepgram token (default TTL 30 s).
- No student names or personal data in TTS text. Raw audio is never stored; the TTS audio cache is purgeable.
- Word times are integer milliseconds from the start of that `speak`'s audio.
- Items marked **[verify day 0]** are not confirmed by the provider docs read on 2026-10-30; Task 8 measures them.

```ts
import type { WordTiming } from "@mento/contracts";

export interface CharTiming {
  char: string;
  start_ms: number;
  end_ms: number;
}

/** ElevenLabs HTTP `with-timestamps` alignment (seconds, parallel arrays). */
export function fromHttpAlignment(a: {
  characters: string[];
  character_start_times_seconds: number[];
  character_end_times_seconds: number[];
}): CharTiming[] {
  return a.characters.map((char, i) => ({
    char,
    start_ms: Math.round((a.character_start_times_seconds[i] ?? 0) * 1000),
    end_ms: Math.round((a.character_end_times_seconds[i] ?? 0) * 1000),
  }));
}

/** ElevenLabs WebSocket alignment (ms, relative to the chunk) shifted to absolute time. */
export function fromStreamAlignment(
  a: { chars: string[]; charStartTimesMs: number[]; charDurationsMs: number[] },
  offset_ms: number,
): CharTiming[] {
  return a.chars.map((char, i) => {
    const start = offset_ms + (a.charStartTimesMs[i] ?? 0);
    return { char, start_ms: Math.round(start), end_ms: Math.round(start + (a.charDurationsMs[i] ?? 0)) };
  });
}

/** Group characters into words (split on whitespace). Punctuation stays attached to the word. */
export function charsToWords(chars: CharTiming[]): WordTiming[] {
  const words: WordTiming[] = [];
  let current: CharTiming[] = [];
  const flush = () => {
    if (current.length === 0) return;
    const first = current[0]!;
    const last = current[current.length - 1]!;
    words.push({ word: current.map((c) => c.char).join(""), start_ms: first.start_ms, end_ms: Math.max(last.end_ms, first.start_ms) });
    current = [];
  };
  for (const c of chars) {
    if (/\s/.test(c.char)) flush();
    else current.push(c);
  }
  flush();
  return words;
}

/** Tutor audio is PCM s16le 24 kHz mono: 48 bytes per millisecond. */
export const PCM_BYTES_PER_MS = (24_000 * 2) / 1000;
export const pcmDurationMs = (bytes: number): number => Math.round(bytes / PCM_BYTES_PER_MS);

export interface StreamAlignmentChunk {
  chars: string[];
  charStartTimesMs: number[];
  charDurationsMs: number[];
}

/** ElevenLabs reports times relative to each chunk: shift by the audio already received. */
export class AlignmentAccumulator {
  private offset_ms = 0;
  private readonly chars: CharTiming[] = [];

  push(alignment: StreamAlignmentChunk | undefined, audioBytes: number): CharTiming[] {
    const out = alignment ? fromStreamAlignment(alignment, this.offset_ms) : [];
    this.chars.push(...out);
    this.offset_ms += pcmDurationMs(audioBytes);
    return out;
  }

  all(): CharTiming[] {
    return this.chars;
  }
}
```

Create `packages/voice/src/server/async-queue.ts`:

```ts
/** Push-style producer, pull-style consumer. Ends with `end()` or rejects with `fail(err)`. */
export class AsyncQueue<T> implements AsyncIterable<T> {
  private items: T[] = [];
  private error: Error | null = null;
  private ended = false;
  private wake: (() => void) | null = null;

  push(item: T): void {
    this.items.push(item);
    this.notify();
  }
  end(): void {
    this.ended = true;
    this.notify();
  }
  fail(error: Error): void {
    this.error = error;
    this.notify();
  }
  private notify(): void {
    const w = this.wake;
    this.wake = null;
    w?.();
  }

  async *[Symbol.asyncIterator](): AsyncGenerator<T> {
    for (;;) {
      if (this.items.length > 0) {
        yield this.items.shift() as T;
        continue;
      }
      if (this.error) throw this.error;
      if (this.ended) return;
      await new Promise<void>((resolve) => (this.wake = resolve));
    }
  }
}
```

```ts
import { VoiceConfigError } from "./errors";

export const DEEPGRAM_LISTEN_URL = "wss://api.deepgram.com/v1/listen";

/** Query parameters the browser appends to the Deepgram WebSocket URL (single source of truth). */
export const DEEPGRAM_LISTEN_PARAMS: Record<string, string> = {
  model: "nova-3",
  language: "pt-BR",
  encoding: "linear16",
  sample_rate: "16000",
  channels: "1",
  interim_results: "true",
  endpointing: "300",
  utterance_end_ms: "1000",
  vad_events: "true",
  smart_format: "true",
  punctuate: "true",
};

export interface MintTokenInput {
  apiKey: string;
  ttlSeconds?: number;
  fetchImpl?: typeof fetch;
}

/** Short-lived Deepgram JWT for the browser (docs: POST /v1/auth/grant, default TTL 30 s). */
export async function mintSttToken(input: MintTokenInput): Promise<{ token: string; expires_in: number }> {
  const { apiKey, ttlSeconds = 30, fetchImpl = fetch } = input;
  if (!apiKey) throw new VoiceConfigError("DEEPGRAM_API_KEY is not set");
  const res = await fetchImpl("https://api.deepgram.com/v1/auth/grant", {
    method: "POST",
    headers: { Authorization: `Token ${apiKey}`, "Content-Type": "application/json" },
    body: JSON.stringify({ ttl_seconds: ttlSeconds }),
  });
  if (!res.ok) throw new Error(`Deepgram token grant failed: HTTP ${res.status}`);
  const body = (await res.json()) as { access_token?: string; expires_in?: number };
  if (!body.access_token) throw new Error("Deepgram token grant returned no access_token");
  return { token: body.access_token, expires_in: body.expires_in ?? ttlSeconds };
}
```

```ts
import { fromHttpAlignment } from "../alignment";
import type { TtsChunk, TtsProvider, TtsRequest } from "./types";

export interface ElevenLabsHttpOptions {
  apiKey: string;
  voiceId: string;
  modelId: string;
  baseUrl?: string;
  fetchImpl?: typeof fetch;
}

/** Simple, reliable path: one request returns the whole audio plus character timings. */
export class ElevenLabsHttpProvider implements TtsProvider {
  constructor(private readonly opts: ElevenLabsHttpOptions) {}

  async *synthesize(req: TtsRequest): AsyncGenerator<TtsChunk> {
    const { apiKey, voiceId, modelId, baseUrl = "https://api.elevenlabs.io", fetchImpl = fetch } = this.opts;
    const body: Record<string, unknown> = { text: req.text, model_id: modelId, language_code: "pt" };
    if (req.previous_text) body.previous_text = req.previous_text;
    if (req.next_text) body.next_text = req.next_text;

    const res = await fetchImpl(`${baseUrl}/v1/text-to-speech/${voiceId}/with-timestamps?output_format=pcm_24000`, {
      method: "POST",
      headers: { "xi-api-key": apiKey, "Content-Type": "application/json" },
      body: JSON.stringify(body),
      signal: req.signal,
    });
    if (!res.ok) throw new Error(`ElevenLabs TTS failed: HTTP ${res.status}`);
    const json = (await res.json()) as {
      audio_base64: string;
      alignment?: { characters: string[]; character_start_times_seconds: number[]; character_end_times_seconds: number[] } | null;
    };
    yield { audio: Buffer.from(json.audio_base64, "base64"), chars: json.alignment ? fromHttpAlignment(json.alignment) : [] };
  }
}
```

```ts
import { createHash } from "node:crypto";
import { mkdir, readFile, writeFile } from "node:fs/promises";
import { join } from "node:path";
import type { WordTiming } from "@mento/contracts";

export interface TtsCacheEntry {
  audio: Buffer;
  words: WordTiming[];
}

/** Disk cache of synthesized speech, keyed by hash(text, voice, model). Purgeable at any time. */
export class TtsCache {
  constructor(private readonly dir: string) {}

  key(parts: { text: string; voiceId: string; modelId: string }): string {
    return createHash("sha256").update(JSON.stringify([parts.text, parts.voiceId, parts.modelId])).digest("hex").slice(0, 32);
  }

  async get(key: string): Promise<TtsCacheEntry | null> {
    try {
      const [audio, meta] = await Promise.all([readFile(join(this.dir, `${key}.pcm`)), readFile(join(this.dir, `${key}.json`), "utf8")]);
      return { audio, words: JSON.parse(meta) as WordTiming[] };
    } catch {
      return null;
    }
  }

  async put(key: string, audio: Buffer, words: WordTiming[]): Promise<void> {
    await mkdir(this.dir, { recursive: true });
    await Promise.all([writeFile(join(this.dir, `${key}.pcm`), audio), writeFile(join(this.dir, `${key}.json`), JSON.stringify(words))]);
  }
}
```

```ts
import WebSocket from "ws";
import { AlignmentAccumulator, type StreamAlignmentChunk } from "../alignment";
import { AsyncQueue } from "../async-queue";
import type { TtsChunk, TtsProvider, TtsRequest } from "./types";

/** The subset of `ws` this provider needs (so tests can inject a fake). */
export interface WsLike {
  readyState: number;
  send(data: string): void;
  close(): void;
  on(event: "open" | "close" | "error" | "message", listener: (...args: any[]) => void): unknown;
  once(event: "open" | "close" | "error", listener: (...args: any[]) => void): unknown;
}
export type WsFactory = (url: string, headers: Record<string, string>) => WsLike;

export interface ElevenLabsWsOptions {
  apiKey: string;
  voiceId: string;
  modelId: string;
  baseUrl?: string; // default wss://api.elevenlabs.io
  factory?: WsFactory;
}

const OPEN = 1;
const defaultFactory: WsFactory = (url, headers) => new WebSocket(url, { headers }) as unknown as WsLike;

/**
 * Streaming TTS. One socket per `speak` (a text message plus an end-of-input message); a spare, already
 * initialized socket is opened ahead of time (`warm()`), so the TLS/WebSocket handshake is paid before it is needed.
 */
export class ElevenLabsWsProvider implements TtsProvider {
  private spare: Promise<WsLike> | null = null;
  private readonly factory: WsFactory;

  constructor(private readonly opts: ElevenLabsWsOptions) {
    this.factory = opts.factory ?? defaultFactory;
  }

  warm(): void {
    if (this.spare) return;
    const p = this.open();
    this.spare = p;
    p.catch(() => {
      if (this.spare === p) this.spare = null;
    });
  }

  close(): void {
    const p = this.spare;
    this.spare = null;
    p?.then((s) => s.close()).catch(() => undefined);
  }

  private url(): string {
    const { voiceId, modelId, baseUrl = "wss://api.elevenlabs.io" } = this.opts;
    const q = new URLSearchParams({
      model_id: modelId,
      output_format: "pcm_24000",
      language_code: "pt",
      auto_mode: "true",
      inactivity_timeout: "60",
    });
    return `${baseUrl}/v1/text-to-speech/${voiceId}/stream-input?${q.toString()}`;
  }

  private open(): Promise<WsLike> {
    return new Promise<WsLike>((resolve, reject) => {
      const socket = this.factory(this.url(), { "xi-api-key": this.opts.apiKey });
      socket.once("open", () => {
        socket.send(JSON.stringify({ text: " " }));
        resolve(socket);
      });
      socket.once("error", (err: Error) => reject(err));
      socket.once("close", () => reject(new Error("ElevenLabs socket closed while connecting")));
    });
  }

  private async take(): Promise<WsLike> {
    let socket: WsLike | null = null;
    if (this.spare) {
      const p = this.spare;
      this.spare = null;
      try {
        const s = await p;
        if (s.readyState === OPEN) socket = s;
      } catch {
        socket = null;
      }
    }
    return socket ?? (await this.open());
  }

  async *synthesize(req: TtsRequest): AsyncGenerator<TtsChunk> {
    if (req.signal?.aborted) return;
    const socket = await this.take();
    this.warm(); // prepare the next one in the background

    const queue = new AsyncQueue<TtsChunk>();
    const acc = new AlignmentAccumulator();
    let finished = false;

    socket.on("message", (data: Buffer) => {
      let msg: { audio?: string | null; alignment?: StreamAlignmentChunk; isFinal?: boolean; error?: string };
      try {
        msg = JSON.parse(data.toString());
      } catch {
        return;
      }
      if (msg.error) {
        finished = true;
        queue.fail(new Error(`ElevenLabs error: ${msg.error}`));
        return;
      }
      if (msg.audio) {
        const audio = Buffer.from(msg.audio, "base64");
        queue.push({ audio, chars: acc.push(msg.alignment, audio.length) });
      }
      if (msg.isFinal) {
        finished = true;
        queue.end();
        socket.close();
      }
    });
    socket.on("error", (err: Error) => {
      finished = true;
      queue.fail(err);
    });
    socket.on("close", () => {
      if (!finished) queue.fail(new Error("ElevenLabs socket closed before the final message"));
    });
    const onAbort = () => {
      finished = true;
      queue.end();
      socket.close();
    };
    req.signal?.addEventListener("abort", onAbort, { once: true });

    socket.send(JSON.stringify({ text: `${req.text} ` }));
    socket.send(JSON.stringify({ text: "" }));

    try {
      for await (const chunk of queue) yield chunk;
    } finally {
      req.signal?.removeEventListener("abort", onAbort);
      if (socket.readyState === OPEN) socket.close();
    }
  }
}
```

```ts
import { charsToWords, type CharTiming } from "../alignment";
import type { TtsCache } from "./cache";
import type { TtsProvider } from "./types";

export interface SpeakResult {
  words: WordTiming[];
  first_byte_ms: number | null;
  audio_bytes: number;
  cached: boolean;
}

export interface StartSpeakRequest {
  action_id: string;
  text: string;
  previous_text?: string;
  next_text?: string;
}

interface Job {
  chunks: Buffer[];
  done: boolean;
  error: Error | null;
  waiters: Set<() => void>;
  abort: AbortController;
}

export interface TtsServiceOptions {
  provider: TtsProvider;
  cache: TtsCache;
  voiceId: string;
  modelId: string;
  /** How long a finished job stays available for GET /api/tts/:action_id. Default 5 minutes. */
  retainMs?: number;
}

/** One job per `speak`: starts synthesizing at once, streams chunks to whoever asks, caches the result. */
export class TtsService {
  private readonly jobs = new Map<string, Job>();
  constructor(private readonly opts: TtsServiceOptions) {}

  has(action_id: string): boolean {
    return this.jobs.has(action_id);
  }

  cancel(action_id: string): void {
    this.jobs.get(action_id)?.abort.abort();
  }

  cancelAll(): void {
    for (const job of this.jobs.values()) job.abort.abort();
    this.opts.provider.close?.();
  }

  start(req: StartSpeakRequest): Promise<SpeakResult> {
    const job: Job = { chunks: [], done: false, error: null, waiters: new Set(), abort: new AbortController() };
    this.jobs.set(req.action_id, job);
    const wake = () => {
      const ws = [...job.waiters];
      job.waiters.clear();
      for (const w of ws) w();
    };
    const t0 = performance.now();

    const run = async (): Promise<SpeakResult> => {
      const { provider, cache, voiceId, modelId } = this.opts;
      const key = cache.key({ text: req.text, voiceId, modelId });
      const hit = await cache.get(key);
      if (hit) {
        job.chunks.push(hit.audio);
        job.done = true;
        wake();
        return { words: hit.words, first_byte_ms: 0, audio_bytes: hit.audio.length, cached: true };
      }
      let first: number | null = null;
      const chars: CharTiming[] = [];
      let bytes = 0;
      try {
        for await (const chunk of provider.synthesize({
          text: req.text,
          previous_text: req.previous_text,
          next_text: req.next_text,
          signal: job.abort.signal,
        })) {
          if (first === null) first = Math.round(performance.now() - t0);
          job.chunks.push(chunk.audio);
          bytes += chunk.audio.length;
          chars.push(...chunk.chars);
          wake();
        }
        const words = charsToWords(chars);
        if (!job.abort.signal.aborted && bytes > 0) await cache.put(key, Buffer.concat(job.chunks), words);
        return { words, first_byte_ms: first, audio_bytes: bytes, cached: false };
      } catch (err) {
        job.error = err instanceof Error ? err : new Error(String(err));
        throw job.error;
      } finally {
        job.done = true;
        wake();
      }
    };

    const result = run();
    result.catch(() => undefined); // errors are reported through the returned promise and stream()
    const retain = setTimeout(() => this.jobs.delete(req.action_id), this.opts.retainMs ?? 5 * 60_000);
    retain.unref?.();
    return result;
  }

  async *stream(action_id: string, signal?: AbortSignal): AsyncGenerator<Buffer> {
    const job = this.jobs.get(action_id);
    if (!job) return;
    let i = 0;
    for (;;) {
      while (i < job.chunks.length) yield job.chunks[i++]!;
      if (job.error) throw job.error;
      if (job.done || signal?.aborted) return;
      await new Promise<void>((resolve) => {
        job.waiters.add(resolve);
        signal?.addEventListener("abort", () => resolve(), { once: true });
      });
    }
  }
}
```

```ts
import type { SttCaption } from "@mento/contracts";

export type SttWord = SttCaption["words"][number];
export type SttEvent = { type: "caption"; caption: SttCaption } | { type: "speech_started" };

interface DgWord {
  word: string;
  punctuated_word?: string;
  start: number;
  end: number;
  confidence: number;
}

const toWord = (w: DgWord): SttWord => ({
  word: w.punctuated_word ?? w.word,
  start_ms: Math.round(w.start * 1000),
  end_ms: Math.round(w.end * 1000),
  confidence: w.confidence,
});

const caption = (kind: SttCaption["kind"], words: SttWord[], speech_final: boolean): SttEvent => ({
  type: "caption",
  caption: { kind, text: words.map((w) => w.word).join(" "), words, speech_final },
});

/**
 * Turns Deepgram live messages into captions. Finalized segments accumulate until the turn ends
 * (`speech_final`, or `UtteranceEnd` as the fallback), so every caption shows the whole utterance so far.
 */
export class DeepgramCaptionBuilder {
  private finals: SttWord[] = [];

  handle(message: unknown): SttEvent[] {
    if (typeof message !== "object" || message === null) return [];
    const m = message as Record<string, unknown>;

    if (m.type === "SpeechStarted") return [{ type: "speech_started" }];

    if (m.type === "UtteranceEnd") {
      if (this.finals.length === 0) return [];
      const words = this.finals;
      this.finals = [];
      return [caption("final", words, true)];
    }

    if (m.type !== "Results") return [];
    const alt = (m.channel as { alternatives?: { words?: DgWord[] }[] } | undefined)?.alternatives?.[0];
    const words = (alt?.words ?? []).map(toWord);
    if (words.length === 0) return [];

    if (m.is_final === true) {
      this.finals.push(...words);
      if (m.speech_final === true) {
        const all = this.finals;
        this.finals = [];
        return [caption("final", all, true)];
      }
      return [caption("interim", [...this.finals], false)];
    }
    return [caption("interim", [...this.finals, ...words], false)];
  }
}
```

```ts
import type { VoiceTokenResponse } from "@mento/contracts";
import { DeepgramCaptionBuilder, type SttEvent } from "./deepgram-parse";

export interface SttSessionOptions {
  /** Calls POST /api/voice/token. */
  getToken: () => Promise<VoiceTokenResponse>;
  onEvent: (event: SttEvent) => void;
  onError?: (error: Error) => void;
  onClose?: () => void;
  WebSocketImpl?: typeof WebSocket;
  /**
   * WebSocket subprotocols that carry the token. Default `["bearer", token]` **[verify day 0]**: the Deepgram
   * docs show `["token", apiKey]` for API keys and do not document the JWT form on the pages read.
   */
  protocolsFor?: (token: string) => string[];
}

export interface SttSession {
  start(): Promise<void>;
  /** 80 ms of 16 kHz mono linear16 audio (1280 samples = 2560 bytes). Ignored while muted. */
  sendAudio(chunk: ArrayBuffer): void;
  /** While muted, audio is dropped and KeepAlive keeps the connection open (it closes after ~10 s without audio). */
  setMuted(muted: boolean): void;
  stop(): void;
}

const KEEPALIVE_MS = 4000;

export function createSttSession(opts: SttSessionOptions): SttSession {
  const WS = opts.WebSocketImpl ?? WebSocket;
  const builder = new DeepgramCaptionBuilder();
  let ws: WebSocket | null = null;
  let muted = false;
  let timer: ReturnType<typeof setInterval> | null = null;

  const stopKeepAlive = () => {
    if (timer) clearInterval(timer);
    timer = null;
  };
  const startKeepAlive = () => {
    stopKeepAlive();
    timer = setInterval(() => ws?.readyState === 1 && ws.send(JSON.stringify({ type: "KeepAlive" })), KEEPALIVE_MS);
  };

  return {
    async start() {
      const t = await opts.getToken();
      const url = `${t.ws_url}?${new URLSearchParams(t.params).toString()}`;
      const protocols = opts.protocolsFor ? opts.protocolsFor(t.token) : ["bearer", t.token];
      ws = new WS(url, protocols);
      const socket = ws;
      socket.onmessage = (e: MessageEvent) => {
        try {
          for (const ev of builder.handle(JSON.parse(String(e.data)))) opts.onEvent(ev);
        } catch (err) {
          opts.onError?.(err instanceof Error ? err : new Error(String(err)));
        }
      };
      socket.onerror = () => opts.onError?.(new Error("Deepgram WebSocket error"));
      socket.onclose = () => {
        stopKeepAlive();
        opts.onClose?.();
      };
      await new Promise<void>((resolve, reject) => {
        socket.onopen = () => resolve();
        const prevError = socket.onerror;
        socket.onerror = (ev) => {
          prevError?.call(socket, ev);
          reject(new Error("Deepgram WebSocket failed to open"));
        };
      });
    },
    sendAudio(chunk) {
      if (!ws || ws.readyState !== 1 || muted) return;
      ws.send(chunk);
    },
    setMuted(value) {
      muted = value;
      if (muted) startKeepAlive();
      else stopKeepAlive();
    },
    stop() {
      stopKeepAlive();
      if (ws && ws.readyState === 1) ws.send(JSON.stringify({ type: "CloseStream" }));
      ws?.close();
      ws = null;
    },
  };
```

```ts
/** PCM signed 16-bit little-endian → Float32 in [-1, 1]. A trailing odd byte is ignored. */
export function pcm16ToFloat32(bytes: Uint8Array): Float32Array {
  const count = Math.floor(bytes.length / 2);
  const view = new DataView(bytes.buffer, bytes.byteOffset, count * 2);
  const out = new Float32Array(count);
  for (let i = 0; i < count; i++) out[i] = view.getInt16(i * 2, true) / 32768;
  return out;
}
```

```ts
const WORKLET = `
class Pcm16kChunker extends AudioWorkletProcessor {
  constructor() { super(); this.buf = new Float32Array(1280); this.n = 0; }
  process(inputs) {
    const ch = inputs[0] && inputs[0][0];
    if (!ch) return true;
    for (let i = 0; i < ch.length; i++) {
      this.buf[this.n++] = ch[i];
      if (this.n === 1280) {
        const out = new Int16Array(1280);
        for (let j = 0; j < 1280; j++) out[j] = Math.max(-1, Math.min(1, this.buf[j])) * 32767;
        this.port.postMessage(out.buffer, [out.buffer]);
        this.n = 0;
      }
    }
    return true;
  }
}
registerProcessor("pcm16k-chunker", Pcm16kChunker);
`;

export interface Mic {
  stop(): void;
  setMuted(muted: boolean): void;

/**
 * Captures the microphone as 80 ms chunks (1280 samples) of 16 kHz mono linear16.
 * The AudioContext runs at 16 kHz so the browser resamples the device stream (tested target: Chrome).
 * Echo cancellation is on, so the tutor's own voice does not trigger SpeechStarted (headphones still recommended).
 */
export async function startMic(onChunk: (pcm: ArrayBuffer) => void): Promise<Mic> {
  const stream = await navigator.mediaDevices.getUserMedia({
    audio: { echoCancellation: true, noiseSuppression: true, channelCount: 1 },
  });
  const ctx = new AudioContext({ sampleRate: 16000 });
  const url = URL.createObjectURL(new Blob([WORKLET], { type: "application/javascript" }));
  await ctx.audioWorklet.addModule(url);
  URL.revokeObjectURL(url);

  const source = ctx.createMediaStreamSource(stream);
  const node = new AudioWorkletNode(ctx, "pcm16k-chunker");
  let muted = false;
  node.port.onmessage = (e: MessageEvent<ArrayBuffer>) => {
    if (!muted) onChunk(e.data);
  };
  source.connect(node);

  return {
    setMuted(value) {
      muted = value;
      stream.getAudioTracks().forEach((t) => (t.enabled = !value));
    },
    stop() {
      node.disconnect();
      source.disconnect();
      stream.getTracks().forEach((t) => t.stop());
      void ctx.close();
    },
  };
}
```

```ts
import type { SpeechPlayer, WordTiming } from "@mento/contracts";
import { pcm16ToFloat32 } from "./pcm";

const SAMPLE_RATE = 24000;

/**
 * Plays the chunked PCM of one `speak` as it downloads (Web Audio), fires word callbacks from the
 * alignment (`setWords`) and exposes the output level for the orb.
 */
export class PcmPlayer implements SpeechPlayer {
  private ctx: AudioContext | null = null;
  private analyser: AnalyserNode | null = null;
  private sources = new Set<AudioBufferSourceNode>();
  private listeners = new Set<(w: WordTiming) => void>();
  private words: WordTiming[] = [];
  private timers: ReturnType<typeof setTimeout>[] = [];
  private controller: AbortController | null = null;

  private ensure(): AudioContext {
    if (!this.ctx) {
      this.ctx = new AudioContext({ sampleRate: SAMPLE_RATE });
      this.analyser = this.ctx.createAnalyser();
      this.analyser.fftSize = 256;
      this.analyser.connect(this.ctx.destination);
    }
    return this.ctx;
  }

  /** Word timings of the audio that is playing (from the `speech_alignment` message); may arrive late. */
  setWords(words: WordTiming[]): void {
    this.words = words;
    if (this.startedAt !== null) this.scheduleWords(this.startedAt);
  }
  private startedAt: number | null = null;

  private scheduleWords(startedAtMs: number): void {
    this.timers.forEach(clearTimeout);
    this.timers = [];
    for (const w of this.words) {
      const delay = Math.max(0, startedAtMs + w.start_ms - performance.now());
      this.timers.push(setTimeout(() => this.listeners.forEach((cb) => cb(w)), delay));
    }
  }

  async play(audio_url: string, signal?: AbortSignal): Promise<void> {
    const ctx = this.ensure();
    await ctx.resume();
    this.stop();
    const controller = new AbortController();
    this.controller = controller;
    signal?.addEventListener("abort", () => controller.abort(), { once: true });

    const res = await fetch(audio_url, { signal: controller.signal });
    if (!res.ok || !res.body) throw new Error(`audio fetch failed: HTTP ${res.status}`);
    const reader = res.body.getReader();

    let nextTime = ctx.currentTime + 0.05;
    this.startedAt = null;
    let leftover = new Uint8Array(0);
    let lastSource: AudioBufferSourceNode | null = null;

    try {
      for (;;) {
        const { done, value } = await reader.read();
        if (done) break;
        const merged = new Uint8Array(leftover.length + value.length);
        merged.set(leftover);
        merged.set(value, leftover.length);
        const usable = merged.length - (merged.length % 2);
        leftover = merged.slice(usable);
        if (usable === 0) continue;

        const samples = pcm16ToFloat32(merged.subarray(0, usable));
        const buffer = ctx.createBuffer(1, samples.length, SAMPLE_RATE);
        buffer.copyToChannel(samples, 0);
        const source = ctx.createBufferSource();
        source.buffer = buffer;
        source.connect(this.analyser!);
        source.start(nextTime);
        if (this.startedAt === null) {
          this.startedAt = performance.now() + (nextTime - ctx.currentTime) * 1000;
          this.scheduleWords(this.startedAt);
        }
        nextTime += buffer.duration;
        this.sources.add(source);
        source.onended = () => this.sources.delete(source);
        lastSource = source;
      }
    } catch (err) {
      if ((err as Error).name === "AbortError") return;
      throw err;
    }

    if (lastSource) await new Promise<void>((resolve) => ((lastSource as AudioBufferSourceNode).onended = () => resolve()));
  }

  stop(): void {
    this.controller?.abort();
    this.controller = null;
    this.timers.forEach(clearTimeout);
    this.timers = [];
    for (const s of this.sources) {
      try {
        s.stop();
      } catch {
        /* already stopped */
      }
    }
    this.sources.clear();
    this.startedAt = null;
  }

  onWord(cb: (word: WordTiming) => void): () => void {
    this.listeners.add(cb);
    return () => this.listeners.delete(cb);
  }

  level(): number {
    if (!this.analyser) return 0;
    const data = new Uint8Array(this.analyser.frequencyBinCount);
    this.analyser.getByteTimeDomainData(data);
    let sum = 0;
    for (const v of data) sum += ((v - 128) / 128) ** 2;
    return Math.min(1, Math.sqrt(sum / data.length) * 3);
  }
}
```

```ts
import { useCallback, useEffect, useRef, useState } from "react";
import type { SttCaption, VoiceTokenResponse } from "@mento/contracts";
import { startMic, type Mic } from "./mic";
import { createSttSession, type SttSession } from "./stt-session";

export interface UseSttOptions {
  /** Calls POST /api/voice/token. */
  getToken: () => Promise<VoiceTokenResponse>;
  /** The student's full utterance (final caption). Send it as a `user_message` with kind "speech". */
  onFinal: (caption: SttCaption) => void;
  /** The student started speaking: stop the tutor's audio and abort the turn. */
  onSpeechStarted: () => void;
}

export type SttState = "idle" | "connecting" | "listening" | "muted" | "error";

export function useStt(opts: UseSttOptions) {
  const [state, setState] = useState<SttState>("idle");
  const [caption, setCaption] = useState<SttCaption | null>(null);
  const session = useRef<SttSession | null>(null);
  const mic = useRef<Mic | null>(null);
  const latest = useRef(opts);
  latest.current = opts;

  const stop = useCallback(() => {
    mic.current?.stop();
    session.current?.stop();
    mic.current = null;
    session.current = null;
    setState("idle");
  }, []);

  const start = useCallback(async () => {
    setState("connecting");
    try {
      const s = createSttSession({
        getToken: () => latest.current.getToken(),
        onEvent: (ev) => {
          if (ev.type === "speech_started") return latest.current.onSpeechStarted();
          setCaption(ev.caption);
          if (ev.caption.kind === "final") latest.current.onFinal(ev.caption);
        },
        onError: () => setState("error"),
        onClose: () => setState((prev) => (prev === "error" ? prev : "idle")),
      });
      await s.start();
      session.current = s;
      mic.current = await startMic((pcm) => s.sendAudio(pcm));
      setState("listening");
    } catch {
      stop();
      setState("error");
    }
  }, [stop]);

  const setMuted = useCallback((muted: boolean) => {
    session.current?.setMuted(muted);
    mic.current?.setMuted(muted);
    setState(muted ? "muted" : "listening");
  }, []);

  useEffect(() => stop, [stop]);

  return { state, caption, start, stop, setMuted };
}
```
